# Minimo Comun Multiplo derivado del GCD
fun lcm(int a, int b) : int {
    int producto = a * b;
    int divisor = gcd(a, b);
    return producto / divisor;
}

# Verifica si n es numero primo
fun esPrimo(int n) : bool {
    if (n <= 1) {
        return false;
    }
    int i = 2;
    while (i * i <= n) {
        if (n % i == 0) {
            return false;
        }
        i++;
    }
    return true;
}

# Calcula base elevado a exp de forma iterativa
fun potencia(int base, int exp) : int {
    if (exp == 0) {
        return 1;
    }
    int resultado = 1;
    int i = 0;
    for (i = 0; i < exp; i++) {
        resultado *= base;
    }
    return resultado;
}

# Calcula el n-esimo termino de Fibonacci iterativamente
fun fibonacci(int n) : int {
    if (n <= 0) {
        return 0;
    }
    if (n == 1) {
        return 1;
    }
    int a = 0;
    int b = 1;
    int i = 2;
    while (i <= n) {
        int c = a + b;
        a = b;
        b = c;
        i++;
    }
    return b;
}

# Clasifica un entero segun su magnitud y signo
fun clasificar(int n) : void {
    if (n < 0) {
        print("negativo");
    } else if (n == 0) {
        print("cero");
    } else if (n < 10) {
        print("un digito");
    } else if (n < 100) {
        print("dos digitos");
    } else {
        print("grande");
    }
}

# Suma acumulada de 1 hasta n usando operador compuesto
fun sumaAcumulada(int n) : int {
    int suma = 0;
    int i = 1;
    while (i <= n) {
        suma += i;
        i++;
    }
    return suma;
}

# Cuenta digitos de un numero entero positivo
fun contarDigitos(int n) : int {
    if (n == 0) {
        return 1;
    }
    int count = 0;
    int x = n;
    while (x > 0) {
        count++;
        x = x / 10;
    }
    return count;
}
