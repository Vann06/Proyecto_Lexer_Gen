objeto Figura {
  area: entero = 0;
  metodo iniciar(inicial: entero) { propio.area = inicial; }
  metodo escalar(factor: entero) { propio.area = factor; }
}

objeto Circulo : Figura {
  radio: entero = 0;
  metodo iniciar(r: entero) { propio.radio = r; }
  metodo romper() {
    propio.noExisteEnElObjeto = 1;
    propio.radio = "no es entero";
  }
}
