-- Cada error de este archivo produce exactamente un diagnostico, con los
-- MISMOS codigos que compiscript.yalp y objetos_es.yalp ante el mismo error
-- conceptual. Prueba que los diagnosticos no dependen de como se llamen los
-- tokens ni las producciones, ni de la FORMA de la sintaxis.

record Punto is
  x : int;
  y : int;
end

class Figura is
  var area : int := 0;
  func init(inicial : int) is
    self.area := inicial;
  end
  func escalar(factor : int) is
    self.area := factor;
  end
end

class Circulo extends Figura is
  var radio : int := 0;
  func init(r : int) is
    self.radio := r;
  end
end

-- Herencia de una clase que nunca se declaro.
class Huerfana extends NoHay is
  var z : int := 0;
end
