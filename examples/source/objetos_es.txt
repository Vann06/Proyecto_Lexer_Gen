# Programa valido: cero diagnosticos semanticos esperados.
# Mismas capacidades que el ejemplo de clases de compiscript, pero
# con una gramatica cuyos tokens y producciones se llaman distinto.

objeto Figura {
  area: entero = 0;

  metodo iniciar(inicial: entero) {
    propio.area = inicial;
  }

  metodo obtenerArea(): entero {
    retorna propio.area;
  }

  metodo escalar(factor: entero) {
    propio.area = factor;
  }
}

objeto Circulo : Figura {
  radio: entero = 0;

  metodo iniciar(r: entero) {
    propio.radio = r;
    propio.area = r + r;
  }

  metodo obtenerRadio(): entero {
    retorna radio;
  }
}

f: Figura = crear Figura(10);
imprime(f.obtenerArea());

c: Circulo = crear Circulo(5);
imprime(c.obtenerRadio());
imprime(c.obtenerArea());
imprime(c.area);

c.radio = 7;
c.area = 100;

# Invocaciones correctas: metodo propio, metodo heredado y funcion libre.
metodo duplicar(n: entero): entero {
  retorna n + n;
}

c.escalar(3);
f.escalar(2);
imprime(duplicar(21));

# Sistema de tipos: inicializador correcto, aritmetica valida y
# asignacion compatible. Esta gramatica no tiene declaracion sin
# anotacion de tipo, asi que no aplica la inferencia.
entera: entero = 1;
otra: entero = 2;
entera = entera + otra;
imprime(entera);

# Retorno tipo-correcto y llamada recursiva con la aridad correcta.
metodo contarAtras(n: entero): entero {
  retorna contarAtras(n);
}

# Registro: el struct de este lenguaje. Mismas capacidades, otros nombres.
registro Punto { x: entero; y: entero; }

origen: Punto = Punto { x: 0, y: 0 };
imprime(origen.x);
origen.y = 5;

# Registro anidado.
registro Caja { esquina: Punto; alto: entero; }
caja: Caja = Caja { esquina: Punto { x: 1, y: 2 }, alto: 10 };
imprime(caja.esquina.y);
